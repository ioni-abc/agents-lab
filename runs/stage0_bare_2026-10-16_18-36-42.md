class ShoppingCart:
    def __init__(self):
        self._items = {}  # name -> {"price": float|int, "quantity": int}

    def add_item(self, name, price, quantity):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("name must be a non-empty string")
        if isinstance(price, bool) or not isinstance(price, (int, float)):
            raise TypeError("price must be a number")
        if price < 0:
            raise ValueError("price cannot be negative")
        if isinstance(quantity, bool) or not isinstance(quantity, int):
            raise TypeError("quantity must be an integer")
        if quantity <= 0:
            raise ValueError("quantity must be positive")

        if name in self._items:
            if self._items[name]["price"] != price:
                raise ValueError(f"{name!r} already in cart with a different price")
            self._items[name]["quantity"] += quantity
        else:
            self._items[name] = {"price": price, "quantity": quantity}

    def total(self):
        return sum(i["price"] * i["quantity"] for i in self._items.values())

    @property
    def items(self):
        return {n: dict(d) for n, d in self._items.items()}

import unittest

class TestShoppingCart(unittest.TestCase):
    def setUp(self):
        self.cart = ShoppingCart()

    def test_empty_total(self):
        self.assertEqual(self.cart.total(), 0)

    def test_add_single_item(self):
        self.cart.add_item("apple", 0.5, 4)
        self.assertEqual(self.cart.items["apple"], {"price": 0.5, "quantity": 4})
        self.assertEqual(self.cart.total(), 2.0)

    def test_same_item_merges_quantity(self):
        self.cart.add_item("apple", 0.5, 2)
        self.cart.add_item("apple", 0.5, 3)
        self.assertEqual(self.cart.items["apple"]["quantity"], 5)

    def test_conflicting_price_rejected(self):
        self.cart.add_item("apple", 0.5, 1)
        with self.assertRaises(ValueError):
            self.cart.add_item("apple", 0.75, 1)

    def test_invalid_inputs(self):
        with self.assertRaises(ValueError):
            self.cart.add_item("", 1, 1)
        with self.assertRaises(ValueError):
            self.cart.add_item("x", -1, 1)
        with self.assertRaises(ValueError):
            self.cart.add_item("x", 1, 0)
        with self.assertRaises(TypeError):
            self.cart.add_item("x", 1, 1.5)
        with self.assertRaises(TypeError):
            self.cart.add_item("x", "1", 1)

if __name__ == "__main__":
